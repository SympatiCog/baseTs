## Phase 2: Core Series Subclass Implementation (3-4 weeks)

### 2.1 TimeSeriesData Class Design
**File:** `baseTs/series.py`

```python
class TimeSeriesData(pd.Series):
    """
    Pandas Series subclass optimized for time series analysis.
    
    Attributes:
        freq: Sampling frequency (Hz)
        signal_name: Name of the signal
        history: Processing history
        metadata: Additional time series metadata
    """
    
    _metadata = ['freq', 'signal_name', 'history', 'is_filtered', 
                 'is_interpolated', 'is_uniform_grid', 'ts_offset',
                 'filtered_indices', 'lowess_fit', 'last_process']
    
    def __init__(self, data=None, index=None, freq=None, **kwargs):
        # Handle various input formats
        if isinstance(data, np.ndarray) and isinstance(index, np.ndarray):
            # Legacy numpy array initialization
            super().__init__(data, index=pd.Index(index))
        elif hasattr(data, 'times') and hasattr(data, 'data'):
            # baseTs object conversion
            super().__init__(data.data, index=pd.Index(data.times))
            self._copy_metadata_from_basetseries(data)
        else:
            super().__init__(data, index=index, **kwargs)
            
        self.freq = freq or self._calculate_effective_frequency()
        self._initialize_metadata(**kwargs)
```

### 2.2 Core Infrastructure Methods
**Essential overrides:**

```python
@property
def _constructor(self):
    return TimeSeriesData

def _constructor_sliced(self):
    return TimeSeriesData

def copy(self, deep=True):
    """Enhanced copy with metadata preservation"""
    copied = super().copy(deep=deep)
    for attr in self._metadata:
        setattr(copied, attr, getattr(self, attr, None))
    return copied

def duration(self) -> float:
    """Calculate time series duration"""
    return float(self.index[-1] - self.index[0])

def len(self) -> int:
    """Backward compatibility method"""
    return len(self)
```

```python
def trimto_timepoints(self, start_time=None, end_time=None, inplace=False):
    """Enhanced with pandas time slicing"""
    if self._backend == 'series':
        # Leverage pandas time-based indexing
        result = self._series.loc[start_time:end_time]
        return self._wrap_result(result, inplace)
    else:
        # Legacy numpy implementation
        return self._legacy_trim(start_time, end_time, inplace)

def resample(self, freq, method='mean'):
    """New pandas-enabled resampling method"""
    if self._backend == 'series':
        resampled = self._series.resample(freq).agg(method)
        return self._wrap_result(resampled, False)
    else:
        raise NotImplementedError("Resampling requires Series backend")
```

**Filter method enhancement:**
```python
def bandpass_at(self, hp_hz=0.01, lp_hz=0.1, inplace=False, **kwargs):
    """Enhanced with metadata preservation"""
    if self._backend == 'series':
        filtered_data = bandpass_filter(self._series.values, hp_hz, lp_hz, 
                                      self._series.freq, **kwargs)
        result = TimeSeriesData(filtered_data, index=self._series.index,
                               freq=self._series.freq)
        result._copy_metadata_from(self._series)
        result.history.append(f"Bandpass filtered at {lp_hz}:{hp_hz} Hz")
        result.is_filtered = True
        return self._wrap_result(result, inplace)
    else:
        # Legacy implementation
        return self._legacy_bandpass_at(hp_hz, lp_hz, inplace, **kwargs)
```

### 3.3 New Enhanced Methods
**Leverage pandas capabilities:**

```python
def align_with(self, other_series, method='outer'):
    """Align two time series on common time index"""
    return self._series.align(other_series, join=method)

def rolling_statistics(self, window, center=True):
    """Rolling statistics with time-aware windows"""
    return self._series.rolling(window, center=center)

def time_slice(self, start=None, end=None):
    """Enhanced time-based slicing"""
    return self._series.loc[start:end]

def interpolate_to_frequency(self, target_freq, method='linear'):
    """Intelligent frequency conversion"""
    new_index = pd.date_range(start=self._series.index[0], 
                             end=self._series.index[-1], 
                             freq=target_freq)
    return self._series.reindex(new_index).interpolate(method=method)
```
